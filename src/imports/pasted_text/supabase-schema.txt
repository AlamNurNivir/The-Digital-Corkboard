3. DATABASE SCHEMA & SUPABASE SETUP (SQL)
----------------------------------------------------------------------
Generate a complete PostgreSQL script for Supabase with Row Level Security (RLS) enabled:

```sql
-- 1. Create Profiles Table
CREATE TABLE public.profiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  avatar_url TEXT DEFAULT '[https://api.dicebear.com/7.x/bottts/svg?seed=default](https://api.dicebear.com/7.x/bottts/svg?seed=default)',
  meme_title TEXT DEFAULT 'Smooth Brain Era 🧠',
  current_streak INT DEFAULT 0,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. Create Learning Logs Table
CREATE TABLE public.learning_logs (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  topic TEXT NOT NULL,
  description TEXT,
  minutes_spent INT NOT NULL CHECK (minutes_spent > 0),
  sticky_color TEXT DEFAULT 'bg-yellow-100',
  date_logged DATE DEFAULT CURRENT_DATE NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Create Sticker Reactions Table (For social interaction on logs)
CREATE TABLE public.log_reactions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  log_id UUID REFERENCES public.learning_logs(id) ON DELETE CASCADE NOT NULL,
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  reaction_type TEXT NOT NULL, -- e.g., '🔥', '🗿', '🧠', '💀', '👏'
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  UNIQUE(log_id, user_id, reaction_type)
);

-- 4. RLS POLICIES
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.learning_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.log_reactions ENABLE ROW LEVEL SECURITY;

-- Profiles: Public read, owner update
CREATE POLICY "Public profiles are viewable by everyone" ON public.profiles FOR SELECT USING (true);
CREATE POLICY "Users can update own profile" ON public.profiles FOR UPDATE USING (auth.uid() = id);

-- Logs: Public read, owner insert/update/delete
CREATE POLICY "Logs viewable by authenticated users" ON public.learning_logs FOR SELECT USING (auth.role() = 'authenticated');
CREATE POLICY "Users can create logs" ON public.learning_logs FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "Users can delete own logs" ON public.learning_logs FOR DELETE USING (auth.uid() = user_id);

-- Reactions: Public read, authenticated create/delete
CREATE POLICY "Reactions viewable by authenticated users" ON public.log_reactions FOR SELECT USING (auth.role() = 'authenticated');
CREATE POLICY "Users can insert reactions" ON public.log_reactions FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "Users can delete own reactions" ON public.log_reactions FOR DELETE USING (auth.uid() = user_id);

-- 5. TRIGGER FOR NEW USER PROFILE CREATION
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$ BEGIN   INSERT INTO public.profiles (id, username, avatar_url)   VALUES (     new.id,     COALESCE(new.raw_user_meta_data->>'username', split_part(new.email, '@', 1)),     '[https://api.dicebear.com/7.x/bottts/svg?seed=](https://api.dicebear.com/7.x/bottts/svg?seed=)' \vert{}\vert{} new.id   );   RETURN NEW; END; $$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
